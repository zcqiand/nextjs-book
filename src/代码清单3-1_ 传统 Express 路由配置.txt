// 传统 Express 路由配置
const express = require('express');
const app = express();
const router = express.Router();
const taskController = require('./controllers/task');
const aboutController = require('./controllers/about');

// 维护一份路由表
router.get('/tasks', taskController.index);      // 任务列表
router.get('/tasks/:taskId', taskController.show); // 任务详情
router.post('/tasks', taskController.create);   // 创建任务
router.get('/about', aboutController.index);   // 关于页面

// 应用中间件
app.use('/', router);
app.listen(3000);
